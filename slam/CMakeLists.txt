cmake_minimum_required(VERSION 3.22)
project(kitti_slam VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 3.4.0 EXACT REQUIRED NO_MODULE)
find_package(GTSAM 4.2.0 EXACT REQUIRED)
find_path(NANOFLANN_INCLUDE_DIR nanoflann.hpp REQUIRED)
add_library(kitti_slam src/geometry.cpp src/cloud.cpp src/io.cpp src/scan_context.cpp src/pose_graph.cpp src/pipeline.cpp)
target_include_directories(kitti_slam PUBLIC include PRIVATE ${NANOFLANN_INCLUDE_DIR})
target_link_libraries(kitti_slam PUBLIC Eigen3::Eigen gtsam)
target_compile_options(kitti_slam PRIVATE -Wall -Wextra -Wpedantic)
add_executable(run_kitti_slam apps/run_kitti_slam.cpp)
target_link_libraries(run_kitti_slam PRIVATE kitti_slam)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest 1.15.2 EXACT REQUIRED)
  add_executable(slam_tests tests/slam_tests.cpp)
  target_link_libraries(slam_tests PRIVATE kitti_slam GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(slam_tests)
endif()
